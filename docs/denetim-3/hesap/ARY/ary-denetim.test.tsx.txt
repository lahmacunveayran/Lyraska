import { act, fireEvent, render, screen, within } from "@testing-library/react";
import { useParametric } from "../hooks/useParametric";
import { useEqualizer } from "../hooks/useEqualizer";
import { ParametricEq } from "./ParametricEq";
import { EqualizerPanel } from "./EqualizerPanel";
import { kindHasGain, previewParametricState } from "../lib/parametric";
import type { ParametricSettings, ParametricState } from "../lib/backend";

type Deferred<T> = { promise: Promise<T>; resolve: (v: T) => void; reject: (e: unknown) => void };
const deferred = <T,>(): Deferred<T> => {
  let resolve!: (v: T) => void;
  let reject!: (e: unknown) => void;
  const promise = new Promise<T>((res, rej) => {
    resolve = res;
    reject = rej;
  });
  return { promise, resolve, reject };
};

/** Çekirdeğin `ParametricSettings::sanitized` işleminin aynısı (peq.rs:649-662). */
const coreSanitize = (s: ParametricSettings): ParametricSettings => ({
  ...s,
  bands: s.bands.slice(0, 20).map((b) => ({
    ...b,
    freqHz: Math.min(24000, Math.max(10, Number.isFinite(b.freqHz) ? b.freqHz : 1000)),
    gainDb: kindHasGain(b.kind) ? Math.min(24, Math.max(-24, b.gainDb)) : 0,
    q: Math.min(30, Math.max(0.1, b.q)),
  })),
});

const core = vi.hoisted(() => ({
  load: null as null | Promise<unknown>,
  sent: [] as unknown[],
  manual: null as null | ((settings: unknown) => Promise<unknown>),
  importResult: null as null | (() => Promise<unknown>),
}));

vi.mock("../lib/backend", async (importOriginal) => {
  const actual = await importOriginal<typeof import("../lib/backend")>();
  return {
    ...actual,
    getParametric: async () => (core.load ? core.load : null),
    setParametric: async (settings: ParametricSettings) => {
      core.sent.push(settings);
      if (core.manual) return core.manual(settings);
      return previewParametricState(coreSanitize(settings));
    },
    setParametricCompare: async () => null,
    pickParametricFile: async () => "C:/eq/bozuk.txt",
    importParametric: async () => (core.importResult ? core.importResult() : null),
  };
});

beforeEach(() => {
  core.load = null;
  core.sent = [];
  core.manual = null;
  core.importResult = null;
});

const lastSent = () => core.sent.at(-1) as ParametricSettings | undefined;

function Harness() {
  const parametric = useParametric();
  return <ParametricEq parametric={parametric} />;
}

const panel = () => screen.getByRole("region", { name: "Parametrik ekolayzer" });

async function mount() {
  await act(async () => {
    render(<Harness />);
  });
  return panel();
}

async function typeInto(field: HTMLElement, value: string) {
  fireEvent.change(field, { target: { value } });
  await act(async () => fireEvent.keyDown(field, { key: "Enter" }));
}

describe("ARY — parametrik ekolayzer arayüzü", () => {
  it("ARY-yerel: Türkçe binlik noktasıyla '2.500' Hz iki bin beş yüz hertz olur (10 Hz değil)", async () => {
    const p = await mount();
    await act(async () => fireEvent.click(within(p).getByRole("button", { name: /^3 · PK/ })));
    await typeInto(within(p).getByRole("textbox", { name: "Frekans" }), "2.500");
    expect(lastSent()?.bands[2]?.freqHz).toBe(2500);
  });

  it("ARY-tür: tür LP'ye çevrilip PK'ye dönünce bandın kazancı kaybolmaz", async () => {
    const p = await mount();
    await act(async () => fireEvent.click(within(p).getByRole("button", { name: /^3 · PK/ })));
    await typeInto(within(p).getByRole("textbox", { name: "Kazanç" }), "+6");
    const kind = () => within(p).getByRole("combobox", { name: "Tür" });
    await act(async () => fireEvent.change(kind(), { target: { value: "lowPass" } }));
    await act(async () => fireEvent.change(kind(), { target: { value: "peaking" } }));
    expect(lastSent()?.bands[2]).toMatchObject({ kind: "peaking", gainDb: 6 });
  });

  it("ARY-duyuru: dosya yükleme hatası ekran okuyucuya duyurulur (role=alert/status/aria-live)", async () => {
    core.importResult = async () => {
      throw '3. satır okunamadı: "Filter 3: ON PK Fc abc Hz"';
    };
    const p = await mount();
    await act(async () => fireEvent.click(within(p).getByRole("button", { name: "Dosyadan yükle" })));
    const message = within(p).getByText(/3\. satır okunamadı/);
    expect(message.closest('[role="alert"],[role="status"],[aria-live]')).not.toBeNull();
  });

  it("ARY-ad: kapalı bandın anahtarı 'açık' diye okunmaz", async () => {
    const p = await mount();
    await act(async () => fireEvent.click(within(p).getByRole("button", { name: /^1 · LS/ })));
    const toggle = within(p).getByRole("switch", { name: /^Bant 1/ });
    await act(async () => fireEvent.click(toggle));
    expect(toggle).toHaveAttribute("aria-checked", "false");
    // Ekran okuyucu: "Bant 1 açık, anahtar, kapalı"
    expect(toggle).not.toHaveAccessibleName("Bant 1 açık");
  });

  it("ARY-odak: düğümde Delete ile son bant silinince odak sayfanın başına düşmez", async () => {
    const p = await mount();
    const node = within(p).getByRole("button", { name: /^Bant 5:/ });
    await act(async () => node.focus());
    await act(async () => fireEvent.keyDown(node, { key: "Delete" }));
    expect(lastSent()?.bands).toHaveLength(4);
    expect(document.activeElement).not.toBe(document.body);
  });

  it("ARY-ölçek: sürüklerken ölçek ±18'den ±24'e geçince kazanç 0,4 piksellik harekette sıçramaz", async () => {
    const p = await mount();
    const plot = p.querySelector(".peq__plot") as HTMLDivElement;
    plot.getBoundingClientRect = () =>
      ({ left: 0, top: 0, width: 1000, height: 360, right: 1000, bottom: 360 }) as DOMRect;
    const node = within(p).getByRole("button", { name: /^Bant 3:/ });
    await act(async () =>
      fireEvent.pointerDown(node, { button: 0, clientX: 500, clientY: 180, pointerId: 1 }),
    );
    const gains: number[] = [];
    for (const y of [11.2, 9.0, 8.6]) {
      await act(async () => fireEvent.pointerMove(node, { clientX: 500, clientY: y, pointerId: 1 }));
      gains.push(lastSent()?.bands[2]?.gainDb ?? NaN);
    }
    console.log(`ARY-ölçek: imleç y = 11,2 / 9,0 / 8,6 px → kazanç ${gains.join(" / ")} dB`);
    expect(Math.abs(gains[2]! - gains[1]!)).toBeLessThan(1);
  });

  it("ARY-yükleme: açılıştaki yükleme bitmeden dokunulursa kayıtlı 8 bant ezilmez", async () => {
    const saved: ParametricSettings = {
      enabled: true,
      design: "rbj",
      bands: Array.from({ length: 8 }, (_, i) => ({
        kind: "peaking" as const,
        freqHz: 100 * (i + 1),
        gainDb: i - 3,
        q: 1,
        enabled: true,
      })),
    };
    const load = deferred<ParametricState>();
    core.load = load.promise;
    const p = await mount();
    // Yükleme sürerken kullanıcı bant ekler.
    await act(async () => fireEvent.click(within(p).getByRole("button", { name: "+ Bant ekle" })));
    await act(async () => load.resolve(previewParametricState(saved)));
    console.log(
      `ARY-yükleme: çekirdeğe giden bant sayısı ${lastSent()?.bands.length} (kayıtlı 8 bant + 1 bekleniyordu)`,
    );
    expect(lastSent()?.bands.length).toBe(9);
  });

  it("ARY-sıra (yeşil beklenir): hızlı değişikliklerde en son ayar kazanır, ara sonuçlar gösterilmez", async () => {
    const replies: Deferred<ParametricState>[] = [];
    core.manual = () => {
      const d = deferred<ParametricState>();
      replies.push(d);
      return d.promise;
    };
    const p = await mount();
    const node = within(p).getByRole("button", { name: /^Bant 2:/ });
    await act(async () => node.focus());
    for (let i = 0; i < 5; i++) {
      await act(async () => fireEvent.keyDown(node, { key: "ArrowUp" }));
    }
    // Yalnızca ilk istek ve en sonuncusu gider.
    expect(core.sent).toHaveLength(1);
    await act(async () =>
      replies[0]!.resolve(previewParametricState(coreSanitize(core.sent[0] as ParametricSettings))),
    );
    expect(core.sent).toHaveLength(2);
    expect((core.sent[1] as ParametricSettings).bands[1]!.gainDb).toBe(2.5);
    await act(async () =>
      replies[1]!.resolve(previewParametricState(coreSanitize(core.sent[1] as ParametricSettings))),
    );
    expect(within(p).getByRole("button", { name: /^Bant 2:.*\+2,5 dB/ })).toBeInTheDocument();
  });
});

describe("ARY — görünüm sekmeleri (WAI-ARIA tabs)", () => {
  function Panel() {
    const equalizer = useEqualizer();
    const parametric = useParametric();
    return <EqualizerPanel equalizer={equalizer} parametric={parametric} />;
  }
  it("ARY-sekme: sekmeler sekme paneline bağlı ve ok tuşlarıyla gezilebilir", async () => {
    await act(async () => {
      render(<Panel />);
    });
    const graphic = screen.getByRole("tab", { name: /Grafik/ });
    expect(graphic).toHaveAttribute("aria-controls");
    await act(async () => graphic.focus());
    await act(async () => fireEvent.keyDown(graphic, { key: "ArrowRight" }));
    expect(document.activeElement).toBe(screen.getByRole("tab", { name: /Parametrik/ }));
  });
});
